PROMPT CREATE OR REPLACE VIEW vwbsnrstbed
CREATE OR REPLACE VIEW vwbsnrstbed (
  loc,
  bedno,
  costid,
  nrstid,
  nrstnm,
  nrstloc,
  icumk,
  sqno,
  bednosts,
  bednostsnm,
  abmk,
  buildingname,
  bedtype,
  bedkd,
  bedkdnm
) AS
SELECT
  LOC, BEDNO, COSTID, NRSTID, NRSTNM, NRSTLOC
  , Decode(ICUNM, '', 'N', 'Y') AS ICUMK  
  , SQNO 
  , BEDNOSTS
  , Decode(BEDNOSTS, '', '', KWBSADM.Get_Bednosts_Desc(BEDNOSTS)) AS BEDNOSTSNM  
  , ABMK
  , Decode(BUILDINGNAME2, '', BUILDINGNAME1, BUILDINGNAME2) AS BUILDINGNAME
  , Decode(ABMK2, '', BEDTYPE1, BEDTYPE2) AS BEDTYPE
  , Decode(ABMK2, '', BEDKD1, BEDKD2) AS BEDKD
  , Decode(ABMK2, '', BEDKDNM1, BEDKDNM2) AS BEDKDNM
FROM (
  SELECT
    A.LOC, A.BEDNO, A.COSTID, NVL(B.ICUNM, A.NRSTID) NRSTID, C.CNM NRSTNM, A.NRSTID NRSTLOC, A.SQNO, A.BEDNOSTS, A.ABMK, B.ICUNM
	, D.BUILDINGNAME AS BUILDINGNAME1
	, DX.BUILDINGNAME AS BUILDINGNAME2
    , E.ABMK AS ABMK1, E.MK2 AS BEDTYPE1, E.KD AS BEDKD1, E.KDNM AS BEDKDNM1
    , EX.ABMK AS ABMK2, EX.MK2 AS BEDTYPE2, EX.KD AS BEDKD2, EX.KDNM AS BEDKDNM2
  FROM RADMBED A, RADMICU B, RASBDEPT C, RADMBUILDINGSTD D, RWBSBUILDINGSTD DX, RADMUBEDSD E, RWBSUBEDSD EX
  WHERE A.COSTID = B.COSTID(+) AND A.COSTID = C.DPID
    AND (A.LOC = D.LOC(+) AND SUBSTR(A.NRSTID,4,1) = D.MK(+) AND A.LOC = DX.LOC(+) AND SUBSTR(A.NRSTID,4,1) = DX.MK(+))
	  AND (A.ABMK = E.ABMK(+) AND A.ABMK = EX.ABMK(+))
    AND A.COSTID IS NOT NULL AND C.ORGNID IN (SELECT DISTINCT ORGNID FROM RSYSLOC)
)
/
